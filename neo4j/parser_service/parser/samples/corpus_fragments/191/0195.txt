#Определяем функцию сохранения 
def SaveToObject(obj, filename): 
 with open(filename, 'wb') as output: 
 pickle.dump(obj, output) 
 
#Опредлеляем функцию загрузки 
def LoadFromObject(filename): 
 f = open(filename, 'rb') 
 loaded_obj = pickle.load(f) 
 f.close() 
 return loaded_obj
